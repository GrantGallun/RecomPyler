"""Narrow output/full-state checks; no compiler imports."""
from pathlib import Path
import struct
import common
import postverify

def sha(path):return common.digest(Path(path).read_bytes())

def parse(path):
    checks,times={},{}
    for line in Path(path).read_text(errors='replace').splitlines():
        f=line.split()
        if len(f)==3 and f[0]=='check':checks[f[1]]=f[2]
        if len(f)==2 and f[0].startswith('seconds'):times[f[0]]=float(f[1])
    return checks,times

def check_equal(a,b):
    return bool(a) and a.keys()==b.keys() and all(struct.pack('!d',float.fromhex(v))==struct.pack('!d',float.fromhex(b[k])) for k,v in a.items())

def audit_compare(reference,unused,data):
    result=postverify.pair(reference,data)
    if not result['valid']:raise ValueError('Full initial/final vectors or intermediate numeric/type/alias checkpoint differs.')
    return {'all_checkpoints_match':True,'checkpoint_comparison':result,'final_type_deltas':{},'initial_type_differences':0}

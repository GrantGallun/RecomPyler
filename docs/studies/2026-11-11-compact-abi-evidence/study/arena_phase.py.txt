"""Task-local arena observer around the frozen repository phase worker."""
import argparse
import functools
import importlib.util
import json
from pathlib import Path
import sys
import types
import time

p = argparse.ArgumentParser(description=__doc__)
p.add_argument('--repo', type=Path, required=True)
p.add_argument('--script', type=Path, required=True)
p.add_argument('--plan', type=Path, required=True)
p.add_argument('--target', required=True)
p.add_argument('--output', type=Path, required=True)
args = p.parse_args()
sys.path.insert(0, str(args.repo / 'src'))
from recompyler import heap, runner
spec = importlib.util.spec_from_file_location('frozen_phase_worker', args.repo / 'tools' / '_phase_worker.py')
worker = importlib.util.module_from_spec(spec)
spec.loader.exec_module(worker)
entries = []
selections=[]
original = heap._Program.compile
original_program_for=heap.Region.program_for
def program_for(region,*values,**named):
    if region.original.__name__!=args.target:return original_program_for(region,*values,**named)
    record={'cached_programs_before':len(region.programs)}
    started=time.perf_counter_ns()
    try:
        return original_program_for(region,*values,**named)
    finally:
        record.update(wall_ns=time.perf_counter_ns()-started,cached_programs_after=len(region.programs))
        selections.append(record)
heap.Region.program_for=program_for
def compile(program, *values, **named):
    result = original(program, *values, **named)
    if program.entry_fn.__name__ != args.target:
        return result
    native = program.run
    @functools.wraps(native)
    def run(*passed, **kw):
        F, I, top, st = passed[:4]
        record = {'capacity_slots':len(F), 'top_before':int(top[0]),
                  'digest':program.digest, 'aot':bool(program.ahead_of_time)}
        try:
            return native(*passed, **kw)
        finally:
            record.update(top_after=int(top[0]), status=int(st[0]))
            entries.append(record)
    program.run = run
    return result
heap._Program.compile = compile
original_types=runner.types
captured=[]
def make_module(*values,**named):
    module=types.ModuleType(*values,**named)
    captured.append(module)
    return module
runner.types=types.SimpleNamespace(ModuleType=make_module)
try:
    result = worker.instrument(dict(repo=str(args.repo.resolve()), script=str(args.script.resolve()),
        plan=str(args.plan.resolve()), function=args.target, line=None, force=True, script_args=['6']))
finally:
    heap._Program.compile = original
    heap.Region.program_for=original_program_for
    runner.types=original_types
result['driver_times_s']=list(captured[-1].__dict__.get('_times',[])) if captured else []
result['driver_times_note']='All six instrumented state-reset kernel times, including external Python energy at unchanged boundary. First cold; remaining five warm. Never headline latency.'
result['program_selection']=selections
result['program_selection_note']='Region.program_for wall includes cold saved-graph/source reconstruction, AOT preparation/validation and any nested load; cached calls retain selection overhead. Overlaps existing phase timers, not an additive new component or pure source-restore timer. No eager seed-module import added.'
result['arena_native_entries'] = entries
result['arena_note'] = 'F/I alias one8byte-per-slot arena; capacity is not live bytes. top includes loaded roots plus retained transient allocations; status2 indicates growth retry. Instrumented, not headline timing.'
result['max_capacity_bytes'] = max((e['capacity_slots']*8 for e in entries), default=0)
result['max_top_slots'] = max((e['top_after'] for e in entries), default=0)
result['growth_native_exits'] = sum(e['status'] == 2 for e in entries)
args.output.parent.mkdir(parents=True, exist_ok=True)
args.output.write_text(json.dumps(result, indent=2)+'\n')
if result.get('error') or not entries or any(not e['aot'] for e in entries) or len(result['driver_times_s'])!=6:
    raise SystemExit('Arena/phase diagnostic failed or used non-AOT path; evidence retained.')

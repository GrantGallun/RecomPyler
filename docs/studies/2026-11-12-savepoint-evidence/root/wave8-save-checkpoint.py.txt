OUT=REPO/'docs/studies/2026-11-12-savepoint-evidence'
read=lambda p:json.loads(p.read_text())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
analysis=ROOT/'wave8-header-analysis'
freeze=read(analysis/'candidate-freeze.json')
assert freeze['commit']=='a9b59be60cdf568c8f6b63442e7efa88fd16fdd2'

full=read(analysis/'full-suite/result.json')

assert sha(analysis/'full-suite/stderr.bin')==full['stderr_sha256']
assert '572 passed, 22 warnings, 52 subtests passed in 336.93s' in (analysis/'full-suite/stdout.bin').read_text()
receipt=read(analysis/'checkpoint-receipt.json')
bundle=Path(receipt['archive'])
assert sha(bundle)==receipt['archive_sha256']=='11430716ee10e137b2f463bd50ab46f1c7b6755b8c9573cb8e64564e500707f8'
with zipfile.ZipFile(bundle) as archive:
    names=archive.namelist()
    assert len(names)==len(set(names))==185

    assert hashlib.sha256(manifest_bytes).hexdigest()==receipt['manifest_sha256']

    assert len(manifest['files_sha256'])==184
    assert set(names)==set(manifest['files_sha256'])|{'metadata/checkpoint-manifest.json'}
    for name,digest in manifest['files_sha256'].items():
        assert hashlib.sha256(archive.read(name)).hexdigest()==digest,name
assert not (ROOT/'wave8-native/results-final').exists()
assert not (ROOT/'wave8-study/results-paired-final').exists()

files=[]


def copy(source,destination):
    source=Path(source);destination=Path(destination)
    if destination.suffix=='.py':destination=destination.with_suffix('.py.txt')
    target=OUT/destination
    assert not target.exists(),target
    target.parent.mkdir(parents=True,exist_ok=True)
    shutil.copyfile(source,target)
    assert source.read_bytes()==target.read_bytes()
    files.append({'source':str(source),'archive':destination.as_posix(),'bytes':target.stat().st_size,'sha256':sha(target)})
